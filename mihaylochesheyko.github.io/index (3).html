Taschenrechner mit Python

Python · Tkinter

Ich habe mit der Programmiersprache Python und der Bibliothek Tkinter einen funktionsfähigen Taschenrechner mit grafischer Oberfläche erstellt. Die Grundstruktur stammt aus einem Tutorial, die Wurzelfunktion (√) habe ich selbst programmiert und zusätzlich die Farben angepasst.

 Zum Tutorial, das ich als Grundlage verwendet habe
Quellcode anzeigen (calculator.py) 
import math
import tkinter

# 1. Button Layout
button_values = [
    ["AC", "+/-", "%", "÷"], 
    ["7", "8", "9", "×"], 
    ["4", "5", "6", "-"],
    ["1", "2", "3", "+"],
    ["0", ".", "√", "="]
]

right_symbols = ["÷", "×", "-", "+", "="]
top_symbols = ["AC", "+/-", "%"]

row_count = len(button_values) # 5
column_count = len(button_values[0]) # 4

# 2. Farben (Lila oben, Blau rechts)
color_purple = "#D8B4FE"
color_black = "#1C1C1C"
color_dark_gray = "#505050"
color_blue = "#007AFF"
color_white = "white"

# 3. Fenster Setup
window = tkinter.Tk()
window.title("Calculator")
window.resizable(False, False)

frame = tkinter.Frame(window)
label = tkinter.Label(frame, text="0", font=("Arial", 45), background=color_black,
                      foreground=color_white, anchor="e", width=column_count)

label.grid(row=0, column=0, columnspan=column_count, sticky="we")

# 4. Buttons erstellen & Farben zuweisen
for row in range(row_count):
    for column in range(column_count):
        value = button_values[row][column]
        button = tkinter.Button(frame, text=value, font=("Arial", 30),
                                width=column_count-1, height=1,
                                command=lambda value=value: button_clicked(value))
        
        if value in top_symbols:
            button.config(foreground=color_black, background=color_purple)
        elif value in right_symbols:
            button.config(foreground=color_white, background=color_blue)
        else:
            button.config(foreground=color_white, background=color_dark_gray)
        
        button.grid(row=row+1, column=column)

frame.pack()

# 5. Logik & Zustand des Taschenrechners
A = "0"
operator = None
B = None

def clear_all():
    global A, B, operator
    A = "0"
    operator = None
    B = None

def remove_zero_decimal(num):
    if num % 1 == 0:
        num = int(num)
    return str(num)

def button_clicked(value):
    global right_symbols, top_symbols, label, A, B, operator

    # Reset, wenn ein Fehler auf dem Display steht
    if label["text"] == "Error":
        label["text"] = "0"

    if value in right_symbols:
        if value == "=":
            if A is not None and operator is not None:
                B = label["text"]
                numA = float(A)
                numB = float(B)

                if operator == "+":
                    label["text"] = remove_zero_decimal(numA + numB)
                elif operator == "-":
                    label["text"] = remove_zero_decimal(numA - numB)
                elif operator == "×":
                    label["text"] = remove_zero_decimal(numA * numB)
                elif operator == "÷":
                    if numB != 0:
                        label["text"] = remove_zero_decimal(numA / numB)
                    else:
                        label["text"] = "Error"
                
                clear_all()

        elif value in "+-×÷":
            if operator is None:
                A = label["text"]
                label["text"] = "0"
                B = "0"
            
            operator = value

    elif value in top_symbols:
        if value == "AC":
            clear_all()
            label["text"] = "0"

        elif value == "+/-":
            result = float(label["text"]) * -1
            label["text"] = remove_zero_decimal(result)

        elif value == "%":
            result = float(label["text"]) / 100
            label["text"] = remove_zero_decimal(result)        

    elif value == "√":
        num = float(label["text"])
        if num < 0:
            label["text"] = "Error"
        else:
            result = math.sqrt(num)
            label["text"] = remove_zero_decimal(result)
        
    else: # Zahlen oder Punkt
        if value == ".":
            if value not in label["text"]:
                label["text"] += value

        elif value in "0123456789":
            if label["text"] == "0":
                label["text"] = value
            else:
                label["text"] += value

# 6. Fenster auf dem Bildschirm zentrieren
window.update()
window_width = window.winfo_width()
window_height = window.winfo_height()
screen_width = window.winfo_screenwidth()
screen_height = window.winfo_screenheight()

window_x = int((screen_width/2) - (window_width/2))
window_y = int((screen_height/2) - (window_height/2))

window.geometry(f"{window_width}x{window_height}+{window_x}+{window_y}")

# 7. App starten
window.mainloop()
